98 void operator () (double, MultiArrayConstView<dcomp,RANK> psi, MultiArrayView<dcomp,RANK> dpsidt) const
99 {
100 if (diagonals.empty()) return;
101
102#ifndef NDEBUG
103 if (psi.extents != calculateAndCheckDimensions(*this)) throw std::runtime_error("Mismatch between StateVector and MultiDiagonal dimensions");
104#endif // NDEBUG
105
106 for (const auto& [index,offsets,diag] : multidiagonal::range(this)) {
107
108 Dimensions ubound{psi.extents}, lboundLHS, lboundRHS;
109 for (auto&& [i,o,u,lL,lR] : std::views::zip(std::views::iota(0uz,RANK),offsets,ubound,lboundLHS,lboundRHS)) {
110 u -= index[i] ? o : 0; lL = index[i] ? 0 : o; lR = index[i] ? o : 0;
111 }
112
113 const auto increment=[&] (auto n, const auto& inc, Dimensions& idxLHS, Dimensions& idxRHS, Dimensions& idxDiag)
114 {
115 using namespace hana::literals;
116 if constexpr (n!=0_c) {
117 if (idxLHS[n]==ubound[n]-1) {idxLHS[n]=lboundLHS[n]; idxRHS[n]=lboundRHS[n]; idxDiag[n]=0; inc(n-1_c,inc,idxLHS,idxRHS,idxDiag);}
118 else {idxLHS[n]++; idxRHS[n]++; idxDiag[n]++;}
119 }
120 else {idxLHS[0]++; idxRHS[0]++; idxDiag[0]++;}
121 };
122
123 for ( Dimensions idxLHS{lboundLHS}, idxRHS{lboundRHS}, idxDiag{}; idxLHS[0]!=ubound[0] ; increment( hana::llong_c<RANK-1>, increment, idxLHS, idxRHS, idxDiag ) )
124 dpsidt(idxLHS) += diag(idxDiag)*psi(idxRHS);
125 }
126 }

149 MultiDiagonal& hermitianConjugate()
150 {
151 Diagonals newDiagonals;
152 for (auto&& [index,offsets,diag] : multidiagonal::range(this)) {
153 Index newIndex{index}; for (size_t i=0; i<RANK; ++i) if (offsets[i]) newIndex.flip(i);
154 conj(diag);
155 newDiagonals[newIndex].emplace(offsets,std::move(diag));
156 }
157 diagonals.swap(newDiagonals);
158 return *this;
159 }

216 friend Dimensions calculateAndCheckDimensions(const MultiDiagonal& md)
217 {
218 if (md.diagonals.empty()) return {};
219
220 auto transformExtentsOfDiagonals = [] (const auto& diagonal) -> Dimensions {
221 auto res{diagonal.second.extents};
222 for (auto&& [v,o] : std::views::zip(res,diagonal.first)) v+=o;
223 return res;
224 };
225
226 auto res{transformExtentsOfDiagonals(*md.diagonals.cbegin()->second.cbegin())};
227
228 if (!std::ranges::fold_left_first( std::views::join(md.diagonals | std::views::values) | std::views::transform([=] (const auto& diagonal) {
229 return std::ranges::equal(res,transformExtentsOfDiagonals(diagonal));
230 }), std::logical_and{} ).value_or(true)) throw std::runtime_error("Dimensions mismatch in MultiDiagonal");
231
232 return res;
233 }
